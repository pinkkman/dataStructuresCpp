void maxim(int arr[], int size,int &max)
{
  max = 1;
  for (int j = 0; j < size; j++)
  {
    if (max < arr[j])
    {
      max = arr[j];
    }
    else
    {
      max = max;
    } }}

void minim(int arr[], int size,int &min)

 min = arr[0];
