#include <iostream>
#include<vector>
using namespace std;
void sorting(int n, vector<int>&arr){
  for(int i=0;i<n-1;i++){
 for(int j=0;j<n-i-1;j++){
if(arr[j]>arr[j+1]){
  swap(arr[j],arr[j+1]);
}
 }
  }
}

int main()
{
vector<int>arr={2,1,3,4};
int n=arr.size();
sorting(n,arr);
for(int i=0;i<n;i++){
  cout<<arr[i];
  cout<<"\t";
}
  return 0;
}
