#include <iostream>
#include<vector>
using namespace std;
 vector<int>pairsum(vector<int>nums,int &tar){
    int size=nums.size();
    vector<int>ans;
    int st=0,end=size-1;
  while(st<end){
    if(nums[st]+nums[end]>tar){
      end--;
    }

 else if(nums[st]+nums[end]<tar){
      st++;
    }
    else{
      ans.push_back(st);
    ans.push_back(end);
     return ans;
    } 
  }return ans;

 }

int main()
{
  //two pointer approach
  int size = 6, tar;
  vector<int>arr = {0,2,5,6,9,16}; //sum_7 =1,2

 tar=7;
 vector<int>ans=pairsum(arr,tar);
 cout<<ans[0]<<" , "<< ans[1];


 
  return 0;
}
