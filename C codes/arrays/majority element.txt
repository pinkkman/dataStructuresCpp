#include <iostream>
#include<vector>
using namespace std;
    int majorityElement(vector<int>& nums) {
int freq=0,ans=0;
for(int value:nums){
if(freq==0){
    ans=value;
} 
if(ans==value){
    freq++;
}else{
freq--;
}
}
if(freq>0){
return ans;
}
else return -1;}
