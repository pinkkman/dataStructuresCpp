#include <iostream>
using namespace std;

int main(){
  int size=6;
int arr[size];
for(int i=0;i<size;i++){
  cout<<"enter the "<<i<<"th number : ";
  cin>>arr[i];
}
int min=arr[0];
for(int j=0;j<size;j++){
 if(min>arr[j]){
  min=arr[j];
 }
 else{
  min=min;

 }
}
cout<<"the smallest array is "<<min;

return 0;
}
